 /* 
  前端HTML程式式调用此方法，传入需要添加的数据 传入数据格式说明
  1.item_type  新增的Item的类型
  2.property_nameo_and_value 新增数据属性名称和相关键值对,例如：{ "item_number": "test123", "name": "测试数据" }
  3. serverevents 新增时候是否触发相关服务器方法默认1触发 若不触发则传入0
 */ 
Aras.Server.Security.Identity plmIdentity = Aras.Server.Security.Identity.GetByName("Aras PLM");//设置权限（getbyname后面的括号填Identity的name）
bool PermissionWasSet = Aras.Server.Security.Permissions.GrantIdentity(plmIdentity);//设置权限
const string ERR_PREFIX = "Error[IC_allReviewPassOrReject]：";
Innovator inn = this.getInnovator();
try 
{
      var hoursReviews = Newtonsoft.Json.JsonConvert.DeserializeObject<List<ICHoursReview>>(this.getProperty("HoursReviewList", "[]"));
      var type = this.getProperty("methodtype", "pass");
      foreach (var hoursReview in hoursReviews)
      {
          var promoteItem = inn.newItem("IC_Working_Hour", "get");
          promoteItem.setProperty("id", hoursReview.id);
          promoteItem.setAttribute("select", "state,id");
          promoteItem = promoteItem.apply();
          if (type == "pass")
          {
              pass(inn, promoteItem);
          }
          else
          {
              reject(inn, promoteItem);
          }
      }
    return this;

} 
catch (Exception ex) 
{
   throw new Exception(ERR_PREFIX + ex.Message);
}
finally 
{
   if (PermissionWasSet) Aras.Server.Security.Permissions.RevokeIdentity(plmIdentity);//释放权限
}}
 /// <summary>
 /// 通过
 /// </summary>
 public  void pass(Innovator inn,Item promoteItem)
 {
     if (promoteItem.getProperty("state", "pass") == "")
     {
         throw new Exception("当前状态为,已通过。无法二次通过请刷新数据");
     }
     if (promoteItem.getProperty("state", "") == "reject")
     {
         throw new Exception("当前状态为已驳回。无法通过请刷新数据");
     }
     var requst = promoteItem.promote("pass", "已通过");
 }
 public  void reject(Innovator inn, Item promoteItem)
 {
     if (promoteItem.getProperty("state", "") == "reject")
     {
         throw new Exception("当前状态为已驳回。无法通过请刷新数据");
     }
     var requst = promoteItem.promote("reject", "已驳回");
 }
